ADD https://ftpmirror.gnu.org/libtasn1/libtasn1-4.21.0.tar.gz /opt/libtasn1.tar.gz
RUN \
	tar -xf libtasn1.tar.gz && \
	cd libtasn1-4.21.0 && \
	./configure --libdir=/usr/local/lib --enable-static --disable-shared && \
	make -j$(nproc) && \
	make install
